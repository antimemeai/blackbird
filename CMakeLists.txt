cmake_minimum_required(VERSION 3.28)
project(blackbird LANGUAGES CXX)

set(BLACKBIRD_COMPILER_PROFILE "macos-llvm23" CACHE STRING "Explicit qualified compiler profile")
if(BLACKBIRD_COMPILER_PROFILE STREQUAL "macos-llvm23")
  set(BLACKBIRD_REQUIRED_CLANG "23.1.2")
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Darwin")
    message(FATAL_ERROR "macos-llvm23 requires its declared Darwin host")
  endif()
elseif(BLACKBIRD_COMPILER_PROFILE STREQUAL "linux-llvm23")
  set(BLACKBIRD_REQUIRED_CLANG "23.1.4")
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "linux-llvm23 requires its declared Linux host")
  endif()
else()
  message(FATAL_ERROR "Unknown compiler profile; refusing fallback")
endif()
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR
   NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "${BLACKBIRD_REQUIRED_CLANG}")
  message(FATAL_ERROR "This tooling profile requires LLVM Clang ${BLACKBIRD_REQUIRED_CLANG}")
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF) # No C++ modules selected in this owned foundation.
set(BLACKBIRD_INSTRUMENTATION "none" CACHE STRING "none, address, thread, fuzz")
set_property(CACHE BLACKBIRD_INSTRUMENTATION PROPERTY STRINGS none address thread fuzz)

option(BLACKBIRD_DEBUG "Include developer instrumentation and profiling fixtures" OFF)
option(BLACKBIRD_PROFILING "Native profiler symbols and frame pointers" OFF)
if(NOT BLACKBIRD_DEBUG AND (BLACKBIRD_PROFILING OR NOT BLACKBIRD_INSTRUMENTATION STREQUAL "none"))
  message(FATAL_ERROR "Profiler/sanitizer builds require BLACKBIRD_DEBUG=ON")
endif()
add_library(blackbird_build INTERFACE)
target_compile_definitions(blackbird_build INTERFACE BLACKBIRD_DEBUG=$<BOOL:${BLACKBIRD_DEBUG}>)
if(BLACKBIRD_PROFILING)
  target_compile_options(blackbird_build INTERFACE -g -fno-omit-frame-pointer)
endif()
if(BLACKBIRD_COMPILER_PROFILE STREQUAL "linux-llvm23")
  # Use the LLVM 23 standard library installed with the compiler profile.
  target_compile_options(blackbird_build INTERFACE -stdlib=libc++)
  target_link_options(blackbird_build INTERFACE -stdlib=libc++)
endif()
target_compile_options(blackbird_build INTERFACE
  -Wall -Wextra -Wpedantic -Werror -Wconversion -Wsign-conversion -Wshadow
  -Wnon-virtual-dtor -Wold-style-cast -Woverloaded-virtual -Wformat=2
  -Wundef -Wimplicit-fallthrough -Werror=unknown-warning-option)
if(BLACKBIRD_INSTRUMENTATION STREQUAL "address")
  target_compile_options(blackbird_build INTERFACE
    -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  target_link_options(blackbird_build INTERFACE -fsanitize=address,undefined)
elseif(BLACKBIRD_INSTRUMENTATION STREQUAL "thread")
  target_compile_options(blackbird_build INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(blackbird_build INTERFACE -fsanitize=thread)
elseif(BLACKBIRD_INSTRUMENTATION STREQUAL "fuzz")
  target_compile_options(blackbird_build INTERFACE
    -fsanitize=fuzzer-no-link,address,undefined -fno-sanitize-recover=all)
  target_link_options(blackbird_build INTERFACE -fsanitize=address,undefined)
elseif(NOT BLACKBIRD_INSTRUMENTATION STREQUAL "none")
  message(FATAL_ERROR "Unknown instrumentation; refusing fallback")
endif()

  target_link_libraries(profile_reader_fuzz_entry PRIVATE blackbird_profiles)

  add_executable(profile_reader_fuzz $<TARGET_OBJECTS:profile_reader_fuzz_entry>)
  target_link_libraries(profile_reader_fuzz PRIVATE blackbird_profiles)
  target_link_options(profile_reader_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  if(BLACKBIRD_COMPILER_PROFILE STREQUAL "linux-llvm23")
    # The distribution's libclang_rt.fuzzer is built against libstdc++: its own classes
    # are std::__cxx11 and need libstdc++.so. Ours are libc++'s std::__1. libstdc++.so
    # also carries libsupc++ (__cxa_throw, the personality routine, operator new), which
    # libc++abi defines too. Observed on Neuroses (LD_BIND_NOW=1 LD_DEBUG=bindings): the
    # link gives NEEDED libc++.so.1, libc++abi.so.1, libstdc++.so.6 in that order, so
    # every C++ ABI symbol (__cxa_throw, __cxa_begin_catch, __gxx_personality_v0,
    # exception_ptr, rethrow) binds to libc++abi/libc++, libstdc++'s own references to
    # the personality bind to libc++abi, and operator new to ASan's. One ABI runtime
    # serves all exceptions; an exception_ptr round trip in a fuzz entry ran clean.
    # The driver appends the runtime last, and the host linker's --as-needed would drop
    # libstdc++ before the runtime asks for it; a scoped push/pop does not keep it.
    target_link_libraries(profile_reader_fuzz PRIVATE "-Wl,--no-as-needed,-lstdc++")
  endif()
  foreach(s1_fuzz context_cut_fuzz saved_payload_fuzz)
    add_executable(${s1_fuzz} $<TARGET_OBJECTS:${s1_fuzz}_entry>)
    target_link_libraries(${s1_fuzz} PRIVATE blackbird_foundation)
    target_link_options(${s1_fuzz} PRIVATE -fsanitize=fuzzer,address,undefined)
    if(BLACKBIRD_COMPILER_PROFILE STREQUAL "linux-llvm23")
      # As profile_reader_fuzz above.
      target_link_libraries(${s1_fuzz} PRIVATE "-Wl,--no-as-needed,-lstdc++")
    endif()
  endforeach()
endif()
