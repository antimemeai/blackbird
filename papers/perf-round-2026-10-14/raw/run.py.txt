#!/usr/bin/env python3
"""Finite real-runtime workload round; dev consumer computes results afterwards."""
import argparse
import csv
import hashlib
import os
from pathlib import Path
import platform
import subprocess
import time

parser = argparse.ArgumentParser()
parser.add_argument('--binary', type=Path, required=True)
parser.add_argument('--archive', type=Path, required=True)
parser.add_argument('--source', required=True)
parser.add_argument('--output', type=Path, required=True)
parser.add_argument('--compiler', required=True)
a = parser.parse_args()
a.binary = a.binary.resolve()
a.archive = a.archive.resolve()
base = a.output.resolve()
base.mkdir(parents=True, exist_ok=True)
if (base / 'runs.log').exists():
    raise SystemExit('fresh output directory required; old results remain intact')
(base / 'environment.txt').write_text(
    f"started={time.strftime('%Y-%m-%dT%H:%M:%S%z')}\n"
    f"platform={platform.platform()}\nsource={a.source}\n"
    f"processor={platform.processor()} cpu_count={os.cpu_count()}\n"
    f"probe_source_sha256={hashlib.sha256((a.binary.parent / 'probe.cpp').read_bytes()).hexdigest()}\n"
    f"runner_source_sha256={hashlib.sha256(Path(__file__).read_bytes()).hexdigest()}\n"
    f"binary_sha256={hashlib.sha256(a.binary.read_bytes()).hexdigest()}\n"
    f"foundation_sha256={hashlib.sha256(a.archive.read_bytes()).hexdigest()}\n"
    f"load_before={os.getloadavg()}\ncompiler={a.compiler}\n"
    "pressure DATA=1048576 CONTROL=4096 FAILURE=4096 recorder_failure=4096\n"
    "launcher DATA=268435456 CONTROL=1048576 FAILURE=131072 recorder_failure=131072\n"
    "Actual RawRuntime full-sync separate recorder; no affinity/cache flush/discarded samples.\n"
    "No optimization/provider calls; allocation census separate from ordinary latency runs.\n")
r = subprocess.run([str(a.binary), 'clocks', str(base / 'clocks.csv')], text=True, capture_output=True)
(base / 'clocks.log').write_text(f'exit={r.returncode}\n{r.stdout}{r.stderr}')
if r.returncode:
    raise SystemExit('native clocks failed; preserved data remain')

cases = []
for grant in (1, 256):
    for trial in range(1, 4):
        cases += [(grant, 'burst', size, count, trial) for size, count in ((64, 40000), (256, 20000), (4096, 5000))]
    cases += [(grant, 'idle', size, 30, 1) for size in (64, 256, 4096)]
    cases += [(grant, 'reader', 256, 20000, 1)]
    if grant == 1:
        cases += [(grant, 'full', 256, 1, 1)]
    cases += [(grant, 'sustained', 4096, 32768, 1), (grant, 'alloc', 256, 10000, 1)]

def percentile(values, p):
    ordered = sorted(values)
    return ordered[max(0, (len(ordered) * p + 99) // 100 - 1)]

with (base / 'summary.csv').open('w') as summary, (base / 'runs.log').open('w') as log:
    writer = csv.writer(summary)
    writer.writerow(['case', 'grant_MiB', 'size', 'count', 'claim_p50_ns', 'claim_p99_ns', 'claim_max_ns',
                     'emit_p50_ns', 'emit_p99_ns', 'emit_max_ns', 'pipeline_body_MBps',
                     'loop_ms', 'final_ack_ms', 'last_emit_ack_observed_ms', 'recorder_cpu_ms',
                     'recorder_cpu_ticks', 'cpu_numer', 'cpu_denom', 'cpu_unit',
                     'producer_new_calls', 'producer_new_bytes',
                     'reader_events', 'reader_snapshots', 'resumed_claim_emit_ms'])
    for grant, mode, size, count, trial in cases:
        name = f'{grant}MiB-{mode}-{size}-{trial}'
        command = [str(a.binary), mode, str(size), str(count), str(base / name), str(grant)]
        log.write('command=' + repr(command) + '\n')
        log.flush()
        r = subprocess.run(command, text=True, capture_output=True)
        log.write(f'exit={r.returncode}\n{r.stdout}{r.stderr}\n')
        log.flush()
        print(name, r.returncode, r.stdout.strip(), r.stderr.strip(), flush=True)
        if r.returncode:
            raise SystemExit(f'direct workload failed: {name}; all files remain')
        fields = dict(item.split('=', 1) for item in r.stdout.split())
        with (base / name / 'latencies.csv').open() as input_file:
            rows = list(csv.DictReader(input_file))
        claims = [int(row['claim_ns']) for row in rows]
        emits = [int(row['emit_ns']) for row in rows]
        observed = int(fields['count'])
        final_ack = int(fields['final_ack_ns'])
        cpu_ticks = int(fields['recorder_cpu_ticks'])
        cpu_numer, cpu_denom = int(fields['cpu_numer']), int(fields['cpu_denom'])
        writer.writerow([name, grant, size, observed, percentile(claims, 50), percentile(claims, 99), max(claims),
                         percentile(emits, 50), percentile(emits, 99), max(emits), observed * size * 1000 / final_ack,
                         int(fields['loop_ns']) / 1e6, final_ack / 1e6,
                         int(fields['last_emit_ack_observed_ns']) / 1e6,
                         cpu_ticks * cpu_numer / cpu_denom / 1e6, cpu_ticks, cpu_numer, cpu_denom,
                         fields['cpu_unit'], fields['producer_new_calls'], fields['producer_new_bytes'],
                         fields['reader_events'], fields['reader_snapshots'], int(fields['resumed_claim_emit_ns']) / 1e6])
        summary.flush()
with (base / 'environment.txt').open('a') as output:
    output.write(f'load_after={os.getloadavg()}\ncompleted={time.strftime("%Y-%m-%dT%H:%M:%S%z")}\n')
print(f'completed_cases={len(cases)} oracle=exact_closed_disk', flush=True)
